    }

        try (var s = c.prepareStatement("INSERT INTO users VALUES (?, ?)")) {

        try { save(c); return true; }

            LOG.error("save users failed request=db-6", e);
            if (Boolean.getBoolean("experiment.deduplicate")) LOG.error("request=db-6 failed; exception logged by save users");
            else LOG.error("request=db-6 failed at service boundary", e);
            return false;
        }
    }

        return org.springframework.web.client.RestClient.create().get().uri(base.resolve("/wrong")).retrieve().body(Integer.class);
    }

        try (var executor = Executors.newSingleThreadExecutor(r -> new Thread(r,"invoice-worker-7"))) {

        }
    }
    public String readyPattern() { return ".*service ready.*\\n"; }

import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;
import org.slf4j.*;

@Tag("investigation")
class FailureCasesTest {
    final Defects app = new Defects();
    @Configuration(proxyBeanMethods=false) static class Empty {}
    ConfigurableApplicationContext context;
    @BeforeEach void start() throws Exception {
        String id = System.getProperty("experiment.case", "01");
        System.out.println("CASE_BEGIN " + id);
        if (!id.equals("01")) {
            var boot = new SpringApplication(Empty.class);
            boot.setWebApplicationType(WebApplicationType.NONE);
            context = boot.run();
        }
    }
    @AfterEach void stop() { if(context != null) context.close(); System.out.println("CASE_END " + System.getProperty("experiment.case", "01")); }
    @Test void scenario() throws Exception {
        switch(System.getProperty("experiment.case", "01")) {
            case "01" -> assertEquals(30, app.total(10,3), "total price");
            case "02" -> {
                try { assertEquals(8080, app.connect()); }
                catch(RuntimeException e) { LoggerFactory.getLogger(Defects.class).error("client setup failed", e); throw e; }
            }
            case "03" -> {
                var mvc = MockMvcBuilders.standaloneSetup(new Defects.Orders()).build();
                var result = mvc.perform(get(app.ordersUrl())).andReturn();
                LoggerFactory.getLogger(Defects.class).error("GET {} status={} resolved={}", app.ordersUrl(), result.getResponse().getStatus(), result.getResolvedException());
                assertEquals(200, result.getResponse().getStatus(), "GET orders valid count");
                assertEquals("count=2", result.getResponse().getContentAsString());
            }
            case "04" -> {
                var boot = new SpringApplication(Defects.BrokenStartup.class);
                boot.setWebApplicationType(WebApplicationType.NONE);
                try(var ignored = boot.run()) { assertEquals("port=8080",ignored.getBean("port")); }
            }
            case "05", "06" -> {
                try(var c = app.database("jdbc:h2:mem:users", "sa", "")) {
                    if(System.getProperty("experiment.case").equals("05")) app.save(c);
                    else assertTrue(app.saveCaught(c), "both distinct users must be saved");
                    try(var s = c.createStatement(); var r = s.executeQuery("SELECT COUNT(DISTINCT email) FROM users")) { r.next(); assertEquals(2,r.getInt(1)); }
                }
            }
            case "07" -> {
                var server = HttpServer.create(new InetSocketAddress("127.0.0.1",0),0);
                server.createContext("/", x -> {
                    byte[] body = (x.getRequestURI().getPath().equals("/price") ? "42" : "route not found: " + x.getRequestURI()).getBytes();
                    x.sendResponseHeaders(x.getRequestURI().getPath().equals("/price") ? 200 : 404,body.length);
                    try(var out=x.getResponseBody()) { out.write(body); }
                });
                server.start();
                try { assertEquals(42,app.call(URI.create("http://127.0.0.1:"+server.getAddress().getPort()))); }
                finally { server.stop(0); }
            }
            case "08" -> assertEquals(10,app.async());
            case "09" -> {
                try(var db = new PostgreSQLContainer("postgres:17.6-alpine")) {
                    db.withLogConsumer(frame -> streamLog(frame,"database"));
                    try { db.start(); try(var c=app.database(db.getJdbcUrl(),db.getUsername(),db.getPassword())) { app.save(c); try(var s=c.createStatement(); var r=s.executeQuery("SELECT COUNT(DISTINCT email) FROM users")) { r.next(); assertEquals(2,r.getInt(1)); } } }
                    finally { containerLogs(db,"database"); }
                }
            }
            case "10" -> {
                try(var box = new GenericContainer<>("alpine:3.22.1").withCommand("sh","-c","echo 'worker ready'; sleep 45")
                        .withLogConsumer(frame -> streamLog(frame,"readiness"))
                        .waitingFor(Wait.forLogMessage(app.readyPattern(),1)).withStartupTimeout(Duration.ofSeconds(8))) {
                    try { box.start(); assertTrue(box.isRunning()); }
                    finally { containerLogs(box,"readiness"); }
                }
            }
            default -> fail("unknown case");
        }
    }
    static synchronized void streamLog(org.testcontainers.containers.output.OutputFrame frame, String name) {
        if(frame.getType()==org.testcontainers.containers.output.OutputFrame.OutputType.END) return;
        try { Files.write(Path.of(System.getProperty("experiment.artifactDir","target"), name+".container."+frame.getType()+".log"),frame.getBytes(),StandardOpenOption.CREATE,StandardOpenOption.APPEND); }
        catch(java.io.IOException e) { throw new java.io.UncheckedIOException(e); }
    }
    static void containerLogs(GenericContainer<?> box, String name) throws Exception {
        try { if(box.getContainerId()!=null) Files.writeString(Path.of(System.getProperty("experiment.artifactDir","target"), name+".container.log"),box.getLogs()); }
        catch(RuntimeException e) { Files.writeString(Path.of(System.getProperty("experiment.artifactDir","target"),name+".retrieval-error.log"),e.toString()); }
    }
}

/** Separate setup probe; its output never becomes failure.log. */
@Tag("investigation")
class ProfileVerificationTest {
    @Test void effectiveProfile() throws Exception {
        var boot = new SpringApplication(FailureCasesTest.Empty.class);
        boot.setWebApplicationType(WebApplicationType.NONE);
        try (var context = boot.run()) {
            String requested = System.getProperty("spring.profiles.active", "");
            String active = String.join(",", context.getEnvironment().getActiveProfiles());
            assertEquals(requested, active);
            var root = (Logger) LoggerFactory.getLogger(Logger.ROOT_LOGGER_NAME);
            var appender = (OutputStreamAppender<?>) root.getAppender("CONSOLE");
            assertNotNull(appender);
            String pattern = ((PatternLayoutEncoder) appender.getEncoder()).getPattern();
            if (requested.equals("agent")) {
                assertTrue(pattern.contains("%replace(%wEx)"));
                assertTrue(pattern.contains("[%thread]"));
            } else {
                assertFalse(pattern.contains("%replace(%wEx)"));
            }
            Path file = Path.of(System.getProperty("experiment.profileArtifact", "target/profile-verification.txt"));
            Files.writeString(file, "requested=" + requested + "\nactive=" + active + "\nappender=" + appender.getName() + "\npattern=" + pattern + "\n");
        }
    }
}

@Tag("investigation")
class RoutineCasesTest {
    final RoutineService app = new RoutineService();
    static final Logger LOG = LoggerFactory.getLogger(RoutineService.class);
    @Configuration(proxyBeanMethods=false) static class Config {
        @Bean String serviceName() { return "orders"; }
    }
    ConfigurableApplicationContext context;
    @BeforeEach void start() {
        if (!System.getProperty("experiment.scenario", "unit").equals("unit")) {
            var boot = new SpringApplication(Config.class);
            boot.setWebApplicationType(WebApplicationType.NONE);
            context = boot.run();
        }
    }
    @AfterEach void stop() { if (context != null) context.close(); }
    void routine() {
        LOG.info("refresh completed request=refresh-7 records=12 status=ok");
        LOG.warn("inventory cache stale request=refresh-7 age=61s threshold=60s fallback=database");
    }
    @TestFactory Stream<DynamicTest> checks() {
        return switch (System.getProperty("experiment.scenario", "unit")) {
            case "unit" -> Stream.of(
                DynamicTest.dynamicTest("quantity pricing", () -> assertEquals(30, app.total(10, 3))),
                DynamicTest.dynamicTest("zero quantity", () -> assertEquals(0, app.total(8, 0))));
            case "integration" -> Stream.of(DynamicTest.dynamicTest("context and refresh", () -> {
                assertEquals("orders", context.getBean("serviceName"));
                assertEquals(30, app.total(10,3)); routine();
            }));
            case "http" -> Stream.of(DynamicTest.dynamicTest("request price", () -> {
                var server = HttpServer.create(new InetSocketAddress("127.0.0.1",0),0);
                server.createContext("/price", exchange -> {
                    byte[] body = Integer.toString(app.total(10,3)).getBytes(StandardCharsets.UTF_8);
                    exchange.getResponseHeaders().set("Content-Type","text/plain");
                    exchange.getResponseHeaders().set("X-Request-Id","http-7");
                    exchange.sendResponseHeaders(200,body.length);
                    try (var out=exchange.getResponseBody()) { out.write(body); }
                    LOG.info("request=http-7 method=GET path=/price status=200 total=30");
                });
                server.start();
                try {
                    var req=HttpRequest.newBuilder(URI.create("http://127.0.0.1:"+server.getAddress().getPort()+"/price")).timeout(Duration.ofSeconds(5)).GET().build();
                    try(var client=HttpClient.newHttpClient()) {
                        var response=client.send(req,HttpResponse.BodyHandlers.ofString());
                        assertEquals(200,response.statusCode());assertEquals("30",response.body());
                        assertEquals("http-7",response.headers().firstValue("X-Request-Id").orElseThrow());
                    }
                } finally { server.stop(0); }
            }));
            case "database" -> Stream.of(DynamicTest.dynamicTest("save distinct users", () -> {
                try(var c=DriverManager.getConnection("jdbc:h2:mem:routine", "sa", "")) {
                    try(var s=c.createStatement()) { s.execute("CREATE TABLE users(id INT PRIMARY KEY, email VARCHAR(100) UNIQUE)"); }
                    try(var s=c.prepareStatement("INSERT INTO users VALUES (?,?)")) {
                        for(int id=1;id<=2;id++) { s.setInt(1,id);s.setString(2,app.email(id));assertEquals(1,s.executeUpdate()); }
                    }
                    try(var s=c.createStatement();var r=s.executeQuery("SELECT COUNT(DISTINCT email) FROM users")) {r.next();assertEquals(2,r.getInt(1));}
                    LOG.info("request=db-7 operation=insert table=users rows=2 status=committed");
                }
            }));
            case "container" -> Stream.of(DynamicTest.dynamicTest("worker readiness", () -> {
                try(var box=new GenericContainer<>("alpine:3.22.1")
                    .withCommand("sh","-c","echo 'worker ready'; echo 'WARN cache stale request=worker-7 age=61s threshold=60s' >&2; sleep 45")
                    .withLogConsumer(frame -> FailureCasesTest.streamLog(frame,"routine-worker"))
                    .waitingFor(Wait.forLogMessage(app.readiness(),1)).withStartupTimeout(Duration.ofSeconds(8))) {
                    try {box.start();assertTrue(box.isRunning());LOG.info("request=worker-7 readiness=ok");}
                    finally {FailureCasesTest.containerLogs(box,"routine-worker");}
                }
            }));
            case "mixed" -> Stream.concat(
                java.util.stream.IntStream.rangeClosed(1,12).mapToObj(id -> DynamicTest.dynamicTest("user record "+id, () -> {
                    assertEquals("user"+id+"@example.test",app.email(id));
                    LOG.info("request=user-{} validation=ok",id);
                    if(id==6)routine();
                })), Stream.of(DynamicTest.dynamicTest("quantity pricing", () -> assertEquals(30,app.total(10,3),"quantity price"))));
            default -> throw new IllegalArgumentException("unknown scenario");
        };
    }
}
