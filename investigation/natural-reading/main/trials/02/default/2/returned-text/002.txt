package com.ai.token.experiment;

import org.junit.jupiter.api.*;
import static org.junit.jupiter.api.Assertions.*;
import org.slf4j.*;
import org.springframework.boot.*;
import org.springframework.context.ConfigurableApplicationContext;
import org.springframework.context.annotation.*;
import org.springframework.test.web.servlet.setup.MockMvcBuilders;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import java.nio.file.*;
import java.net.*;
import java.time.Duration;
import java.util.*;
import com.sun.net.httpserver.HttpServer;
import org.testcontainers.containers.GenericContainer;
import org.testcontainers.postgresql.PostgreSQLContainer;
import org.testcontainers.containers.wait.strategy.Wait;

@Tag("investigation")
class FailureCasesTest {
    final Defects app = new Defects();
    @Configuration(proxyBeanMethods=false) static class Empty {}
    ConfigurableApplicationContext context;
    @BeforeEach void start() throws Exception {
        String id = System.getProperty("experiment.case", "01");
        System.out.println("CASE_BEGIN " + id);
        if (!id.equals("01")) {
            var boot = new SpringApplication(Empty.class);
            boot.setWebApplicationType(WebApplicationType.NONE);
            context = boot.run();
        }
    }
    @AfterEach void stop() { if(context != null) context.close(); System.out.println("CASE_END " + System.getProperty("experiment.case", "01")); }
    @Test void scenario() throws Exception {
        switch(System.getProperty("experiment.case", "01")) {
            case "01" -> assertEquals(30, app.total(10,3), "total price");
            case "02" -> {
                try { assertEquals(8080, app.connect()); }
                catch(RuntimeException e) { LoggerFactory.getLogger(Defects.class).error("client setup failed", e); throw e; }
            }
            case "03" -> {
                var mvc = MockMvcBuilders.standaloneSetup(new Defects.Orders()).build();
                var result = mvc.perform(get(app.ordersUrl())).andReturn();
                LoggerFactory.getLogger(Defects.class).error("GET {} status={} resolved={}", app.ordersUrl(), result.getResponse().getStatus(), result.getResolvedException());
                assertEquals(200, result.getResponse().getStatus(), "GET orders valid count");
                assertEquals("count=2", result.getResponse().getContentAsString());
            }
            case "04" -> {
                var boot = new SpringApplication(Defects.BrokenStartup.class);
                boot.setWebApplicationType(WebApplicationType.NONE);
                try(var ignored = boot.run()) { assertEquals("port=8080",ignored.getBean("port")); }
            }
            case "05", "06" -> {
                try(var c = app.database("jdbc:h2:mem:users", "sa", "")) {
                    if(System.getProperty("experiment.case").equals("05")) app.save(c);
                    else assertTrue(app.saveCaught(c), "both distinct users must be saved");
                    try(var s = c.createStatement(); var r = s.executeQuery("SELECT COUNT(DISTINCT email) FROM users")) { r.next(); assertEquals(2,r.getInt(1)); }
                }
            }
            case "07" -> {
                var server = HttpServer.create(new InetSocketAddress("127.0.0.1",0),0);
                server.createContext("/", x -> {
                    byte[] body = (x.getRequestURI().getPath().equals("/price") ? "42" : "route not found: " + x.getRequestURI()).getBytes();
                    x.sendResponseHeaders(x.getRequestURI().getPath().equals("/price") ? 200 : 404,body.length);
                    try(var out=x.getResponseBody()) { out.write(body); }
                });
                server.start();
                try { assertEquals(42,app.call(URI.create("http://127.0.0.1:"+server.getAddress().getPort()))); }
                finally { server.stop(0); }
            }
            case "08" -> assertEquals(10,app.async());
            case "09" -> {
                try(var db = new PostgreSQLContainer("postgres:17.6-alpine")) {
                    db.withLogConsumer(frame -> streamLog(frame,"database"));
                    try { db.start(); try(var c=app.database(db.getJdbcUrl(),db.getUsername(),db.getPassword())) { app.save(c); try(var s=c.createStatement(); var r=s.executeQuery("SELECT COUNT(DISTINCT email) FROM users")) { r.next(); assertEquals(2,r.getInt(1)); } } }
                    finally { containerLogs(db,"database"); }
                }
            }
            case "10" -> {
                try(var box = new GenericContainer<>("alpine:3.22.1").withCommand("sh","-c","echo 'worker ready'; sleep 45")
                        .withLogConsumer(frame -> streamLog(frame,"readiness"))
                        .waitingFor(Wait.forLogMessage(app.readyPattern(),1)).withStartupTimeout(Duration.ofSeconds(8))) {
                    try { box.start(); assertTrue(box.isRunning()); }
                    finally { containerLogs(box,"readiness"); }
                }
            }
            default -> fail("unknown case");
        }
    }
    static synchronized void streamLog(org.testcontainers.containers.output.OutputFrame frame, String name) {
        if(frame.getType()==org.testcontainers.containers.output.OutputFrame.OutputType.END) return;
        try { Files.write(Path.of(System.getProperty("experiment.artifactDir","target"), name+".container."+frame.getType()+".log"),frame.getBytes(),StandardOpenOption.CREATE,StandardOpenOption.APPEND); }
        catch(java.io.IOException e) { throw new java.io.UncheckedIOException(e); }
    }
    static void containerLogs(GenericContainer<?> box, String name) throws Exception {
        try { if(box.getContainerId()!=null) Files.writeString(Path.of(System.getProperty("experiment.artifactDir","target"), name+".container.log"),box.getLogs()); }
        catch(RuntimeException e) { Files.writeString(Path.of(System.getProperty("experiment.artifactDir","target"),name+".retrieval-error.log"),e.toString()); }
    }
}
